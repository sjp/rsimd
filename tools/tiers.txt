# rsimd ISA tier table: the single source of truth for tier build settings.
#
# Read by ./configure (and configure.win) at install time and by the
# maintainer scripts in tools/. It ships in the package tarball.
# One row per (tier, arch) pair. Fields are separated by '|'; surrounding
# whitespace is ignored. Lines starting with '#' and blank lines are skipped.
#
#   tier     tier id as used in R, C and file names
#   arch     target the row applies to: all, x86_64, i686, aarch64, armv7
#   os       operating systems the row applies to, comma separated:
#              any, or a list of linux, darwin, windows, freebsd
#   flags    extra compiler flags for the tier's translation unit ('-' = none)
#   headers  headers the tier's probe includes, space separated ('-' = none):
#              x86/...      a SIMDe header under src/vendor/simde
#              <...>        a system header
#              @avx512      the curated x86/avx512/*.h list in tools/simde_probe.c
#   enabled  yes, or no for reserved tiers that have no kernels
#
# A tier may have several rows for the same arch: configure uses the first row
# whose probe compiles (sve2 falls back to an older -march spelling).
#
# Tier-specific flags such as -mavx2 apply only to that tier's translation
# unit, which is entered only after a runtime CPU check. No row may use
# -march=native.
#
# tier   | arch    | os            | flags                                            | headers                          | enabled
none     | all     | any           | -                                                | <math.h>                         | yes
sse2     | x86_64  | any           | -                                                | x86/sse2.h x86/sse4.2.h          | yes
sse2     | i686    | any           | -msse2 -mfpmath=sse                              | x86/sse2.h x86/sse4.2.h          | yes
avx2     | x86_64  | any           | -mavx2 -mfma                                     | x86/avx2.h x86/fma.h             | yes
avx512   | x86_64  | any           | -mavx512f -mavx512bw -mavx512dq -mavx512vl -mfma | @avx512                          | yes
neon     | aarch64 | any           | -                                                | x86/sse2.h x86/sse4.2.h x86/fma.h | yes
neon     | armv7   | linux,freebsd | -mfpu=neon                                       | x86/sse2.h x86/sse4.2.h          | yes
sve      | aarch64 | linux,freebsd | -march=armv8-a+sve                               | <arm_sve.h>                      | yes
sve2     | aarch64 | linux,freebsd | -march=armv9-a+sve2                              | <arm_sve.h>                      | yes
sve2     | aarch64 | linux,freebsd | -march=armv8.5-a+sve2                            | <arm_sve.h>                      | yes
rvv      | -       | -             | -                                                | -                                | no
wasm128  | -       | -             | -                                                | -                                | no
